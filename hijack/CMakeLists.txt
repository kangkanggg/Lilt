cmake_minimum_required(VERSION 3.13)
project(Lilt VERSION 0.1.0 LANGUAGES C CXX)
option(ENABLE_DEBUG "Enable debug build flags" OFF)
add_compile_definitions(_GNU_SOURCE)
add_compile_options(-Wall -Wshadow -Werror -Wno-format)
if(ENABLE_DEBUG)
    add_compile_options(-g -O0)
else()
    add_compile_options(-g -O2)
endif()
find_package(Threads REQUIRED)
set(LILT_COMMON_SOURCES
    src/cuda_originals.c src/activity_tracker.c src/graph_scheduler.c
    src/nvml_entry.c src/driver_proxy_compat.c src/loader.c)
foreach(role hp be)
    add_library(lilt-${role} SHARED src/${role}_intercept.c ${LILT_COMMON_SOURCES})
    target_compile_features(lilt-${role} PRIVATE c_std_11)
    target_link_libraries(lilt-${role} PRIVATE Threads::Threads ${CMAKE_DL_LIBS} rt m)
endforeach()
target_compile_definitions(lilt-hp PRIVATE LILT_PROXY_HP=1)
target_compile_definitions(lilt-be PRIVATE LILT_PROXY_BE=1)
